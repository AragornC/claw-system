1import pandas as pd
2import numpy as np
3import talib.abstract as ta
4
5def compute_feature(
6 df: pd.DataFrame,
7 period: int = 14,
8 threshold: float = 0.02
9) -> pd.Series:
10 # 用 ATR 归一化衡量波动幅度
11 atr = ta.ATR(df, timeperiod=max(2, int(period)))
12 close = df['close'].replace(0, np.nan)
13 volatility = (atr / close).clip(lower=0.0)
14
15 # 稳定度 = 1 - 归一化波动率，clamp 到 [0, 1]
16 stability = (1.0 - volatility / threshold).clip(0.0, 1.0)
17 return stability.reindex(df.index).fillna(0.0)
